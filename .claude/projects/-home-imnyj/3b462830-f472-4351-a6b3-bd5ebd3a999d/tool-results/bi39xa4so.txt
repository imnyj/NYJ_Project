    77	def normalize_model_name(name: Any) -> str:
    78	    """Resolve an alias, or a model class, to its canonical registry name."""
    79	    if not isinstance(name, str):
    80	        return getattr(name, "__name__", None) or type(name).__name__
    81	    clean = name.replace("-", "").replace("_", "").lower()
    82	    return _CANONICAL_BY_CLEAN.get(clean, name)

    85	# ----------------------------------------------------------------------------
    86	# Reward-weight search space (Conversation.md section 3, line 31)
    87	#
    88	#   R_t = -( w1*Norm(e_t^2) + w2*Norm(P_tx) + w3*Norm(C_freq) + w4*I_redundant )
    89	#
    90	# The design mandates that w1..w4 are NOT fixed heuristically but are part of the
    91	# Optuna search space so the agent finds its own reward balance. Every term is
    92	# already Min-Max normalised to [0, 1] by the environment, so only the RELATIVE
    93	# balance is meaningful; the sampled raw weights are therefore normalised to sum
    94	# to 1.0 before being handed to the environment. That keeps the reward magnitude
    95	# comparable across trials (so lr / entropy_coef stay on a comparable scale) and
    96	# removes any degenerate incentive to shrink all four weights at once.
    97	# The design defaults 0.5 / 0.2 / 0.2 / 0.1 lie inside these ranges.
    98	# ----------------------------------------------------------------------------
    99	# REWARD_WEIGHT_KEYS is owned by `src/hot_swap_trainer.py` and imported above so
   100	# that the search space, the training env and the evaluation env cannot drift.
   101	
   102	# Tuning seeds, disjoint from the training seeds (42..141, i.e. base 42 plus the
   103	# episode index) and from the evaluation seeds (5001..5005 in `src/evaluate.py`).
   104	# Selecting hyperparameters on a traffic realisation the models are later scored
   105	# on is the ordinary form of leakage in this pipeline, and it was present until
   106	# 2026-09-01: HPO, training and evaluation all shared seed 42.
   107	HPO_TUNING_SEEDS: Tuple[int, ...] = (1001, 1002, 1003)
   108	
   109	REWARD_WEIGHT_RANGES: Dict[str, Tuple[float, float]] = {
   110	    "w1": (0.10, 1.00),   # estimation-error penalty  Norm(e_t^2)
   111	    "w2": (0.02, 0.60),   # transmit-power penalty    Norm(P_tx)
   112	    "w3": (0.02, 0.60),   # channel-congestion penalty Norm(C_freq)
   113	    "w4": (0.02, 0.60),   # redundant-update penalty  I_redundant
   114	}

   117	def sample_reward_weights(trial: optuna.Trial) -> Dict[str, float]:
   118	    """Samples the four reward weights w1..w4 and normalises them to sum to 1.0.
   119	
   120	    The raw samples are registered as Optuna parameters `w1_raw`..`w4_raw`; the
   121	    normalised values that the environment actually trains under are recorded as
   122	    trial user attributes `w1`..`w4` so they are exported to CSV.
   123	    """
   124	    raw: Dict[str, float] = {}
   125	    for key in REWARD_WEIGHT_KEYS:
   126	        lo, hi = REWARD_WEIGHT_RANGES[key]
   127	        raw[key] = trial.suggest_float(f"{key}_raw", lo, hi, log=True)
   128	
   129	    total = sum(raw.values())
   130	    if total <= 0.0:
   131	        total = 1.0
   132	    weights = {k: round(float(v / total), 6) for k, v in raw.items()}
   133	
   134	    for k, v in weights.items():
   135	        trial.set_user_attr(k, v)
   136	    return weights

   292	def assert_hparams_reach_model(model_name: str) -> List[str]:
   293	    """Names this model's search space samples that its constructor cannot accept.
   294	
   295	    An empty list means every sampled key is a real named argument. A non-empty
   296	    one means those keys would vanish into `**hparams` and the trial would report
   297	    a tuning result it never actually applied.
   298	    """
   299	    import inspect
   300	
   301	    canonical_name = normalize_model_name(model_name)
   302	    sampled = set(_search_space_keys(canonical_name))
   303	    accepted = set(inspect.signature(get_baseline(canonical_name).__init__).parameters)
   304	    return sorted(sampled - accepted)

   307	def _search_space_keys(canonical_name: str) -> List[str]:
   308	    """The keys `sample_hparams` would sample, without needing a live trial."""
   309	    class _KeyRecorder:
   310	        def __init__(self) -> None:
   311	            self.keys: List[str] = []
   312	
   313	        def suggest_float(self, name, *a, **k):
   314	            self.keys.append(name)
   315	            return 0.0
   316	
   317	        def suggest_categorical(self, name, choices, *a, **k):
   318	            self.keys.append(name)
   319	            return choices[0]
   320	
   321	        def suggest_int(self, name, *a, **k):
   322	            self.keys.append(name)
   323	            return 0
   324	
   325	    rec = _KeyRecorder()
   326	    sample_hparams(rec, canonical_name)  # type: ignore[arg-type]
   327	    return rec.keys

   330	#: Score handed to a trial whose rollout measured nothing at all. It has to be
   331	#: far above any attainable real score (committed studies sit in 0.887..1.459)
   332	#: because the study minimises and an unmeasured run scores 0.0 on every term.
   333	FAILED_RUN_PENALTY: float = 100.0
   334	
   335	
   336	def run_is_empty(metrics: Dict[str, Any]) -> bool:
   337	    """True when this rollout observed nothing, so its metrics mean nothing.
   338	
   339	    `AoiV2IEnv.get_metrics` degrades gracefully when no vehicle was ever in
   340	    range: mean_aoi 0.0, mean_error 0.0, packet_loss 0/max(1,0) = 0.0 and the
   341	    power falling back to the decoder floor, which normalises to 0.0. Fed to the
   342	    composite that is a score of exactly 0.0 -- the global minimum of a
   343	    minimisation study. Any hyperparameter combination that happens to kill the
   344	    environment would therefore be selected as "optimal". `n_observations` and
   345	    `tx_attempts` are the two signals that say the run was empty; both are
   346	    already produced by `get_metrics` and were simply never read here.
   347	
   348	    Keys that are absent are not treated as zero: callers that score a
   349	    pre-aggregated metric dict are not claiming anything about emptiness.
   350	    """
   351	    if metrics.get("run_failed"):
   352	        return True
   353	    for key in ("n_observations", "tx_attempts"):
   354	        val = metrics.get(key, None)
   355	        if val is not None and float(val) <= 0.0:
   356	            return True
   357	    return False

   360	def compute_composite_objective(
   361	    metrics: Dict[str, float],
   362	    w_error: float = 1.0,
   363	    w_aoi: float = 0.5,
   364	    w_outage: float = 2.0,
   365	    w_power: float = 0.2,
   366	) -> float:
   367	    """
   368	    Formulates the composite objective function balancing:
   369	    1. Tracking estimation error (m)
   370	    2. Age of Information (s)
   371	    3. Outage rate (packet loss ratio in [0, 1])
   372	    4. Normalized transmission power (in [0, 1] mapped from [P_MIN, P_MAX] dBm)
   373	
   374	    A run that measured nothing is scored `FAILED_RUN_PENALTY`, never 0.0.
   375	    """
   376	    if run_is_empty(metrics):
   377	        return float(FAILED_RUN_PENALTY)
   378	
   379	    mean_err = metrics.get("mean_error", 0.0)
   380	    mean_aoi = metrics.get("mean_aoi", 0.0)
   381	    outage_rate = metrics.get("outage_rate", metrics.get("packet_loss_rate", 0.0))
   382	    power_norm = metrics.get("avg_power_norm", 0.5)
   383	
   384	    objective_val = (
   385	        w_error * mean_err
   386	        + w_aoi * mean_aoi
   387	        + w_outage * outage_rate
   388	        + w_power * power_norm
   389	    )
   390	    return float(objective_val)

   393	def evaluate_model_in_env(
   394	    model: Any,
   395	    seed: int = 42,
   396	    n_steps: int = 35,
   397	    density: float = 25.0,
   398	    rsu_range: float = DEFAULT_RSU_RANGE,
   399	    train_steps_during_rollout: int = 0,
   400	    reward_weights: Optional[Dict[str, float]] = None,
   401	) -> Dict[str, float]:
   402	    """
   403	    Evaluates a baseline model in the genuine SUMO AoiV2IEnv for a specific seed.
   404	    Runs real vehicle kinematics, TraCI TLS phase transitions, hybrid action scheduling,
   405	    Rayleigh-SINR co-channel contention, retrospective position estimation error, and AoI.
   406	
   407	    `reward_weights` are the sampled w1..w4 of the design reward; when given they are
   408	    passed to the environment constructor so the trial actually trains under them.
   409	
   410	    The vehicle population is set by `density`; the `n_vehicles` and `rsu_pos`
   411	    arguments this function used to accept were never read, and the tests that
   412	    passed them believed they were controlling something they were not.
   413	    """
   414	    effective_weights = reward_weights if reward_weights else DEFAULT_REWARD_WEIGHTS
   415	    env_kwargs: Dict[str, Any] = {
   416	        key: float(effective_weights[key])
   417	        for key in REWARD_WEIGHT_KEYS
   418	        if key in effective_weights
   419	    }
   420	
   421	    env = AoiV2IEnv(
   422	        density=density,
   423	        seed=seed,
   424	        max_steps=n_steps,
   425	        rsu_range=rsu_range,
   426	        warmup_steps=DEFAULT_WARMUP_STEPS,
   427	        **env_kwargs,
   428	    )
   429	    obs, info = env.reset()
   430	    # The uplink success draw uses the process-wide `random` stream, and SUMO
   431	    # scenario generation consumes a variable number of draws from it before we
   432	    # get here. Re-seeding after reset() pins the simulation stream to
   433	    # (density, seed) whatever ran before.
   434	    random.seed(seed)
   435	    buffer = RetrospectiveReplayBuffer(capacity=1000) if train_steps_during_rollout > 0 else None
   436	    n_update_failures = 0
   437	
   438	    # Event-driven SMDP rollout, matching run_hot_swap_training. Granting every
   439	    # vehicle on every step -- what this loop used to do -- makes Delta inert, so
   440	    # every trial would have measured the same fixed transmit schedule and Optuna
   441	    # would have optimised hyperparameters against a constant.
   442	    open_decision: Dict[str, Dict[str, Any]] = {}
   443	    action_dict: Dict[str, Any] = {}
   444	    for vid, s_vec in obs.items():
   445	        grant, raw_action, _ = model.select_action(s_vec, deterministic=False)
   446	        open_decision[vid] = {"state": np.asarray(s_vec, dtype=np.float32), "raw_action": raw_action}
   447	        action_dict[vid] = grant
   448	
   449	    for step in range(n_steps):
   450	        next_obs, rewards, terminateds, truncateds, step_info = env.step(action_dict)
   451	
   452	        # Close the intervals that finished, then re-decide for whoever needs it.
   453	        if buffer is not None:
   454	            for rec in step_info["completed"]:
   455	                vid = rec["vid"]
   456	                prev = open_decision.get(vid)
   457	                if prev is None:
   458	                    continue
   459	                s2 = next_obs.get(vid)
   460	                done = rec["done"] if s2 is not None else True
   461	                if s2 is None:
   462	                    s2 = np.zeros(STATE_DIM, dtype=np.float32)
   463	                buffer.push(prev["state"], prev["raw_action"], rec["reward"],
   464	                            np.asarray(s2, dtype=np.float32), bool(done), rec["delta_actual"])
   465	
   466	            if len(buffer) >= 16 and train_steps_during_rollout > 0:
   467	                for _ in range(train_steps_during_rollout):
   468	                    batch = buffer.sample(batch_size=16)
   469	                    try:
   470	                        model.update(batch)
   471	                    except Exception:
   472	                        # This used to be `pass`. A baseline whose update always
   473	                        # raised (a batch-format mismatch, say) would finish HPO
   474	                        # normally and report "tuned" hyperparameters for a model
   475	                        # that never took a single gradient step. The count is
   476	                        # returned and recorded as a trial user attribute.
   477	                        n_update_failures += 1
   478	                        if n_update_failures == 1:
   479	                            logger.exception(
   480	                                "%s: model.update() raised on the first training batch; "
   481	                                "further failures in this rollout are counted, not logged.",
   482	                                type(model).__name__,
   483	                            )
   484	
   485	        for rec in step_info["completed"]:
   486	            open_decision.pop(rec["vid"], None)
   487	
   488	        action_dict = {}
   489	        for vid in step_info["needs_decision"]:
   490	            s_vec = next_obs.get(vid)
   491	            if s_vec is None:
   492	                continue
   493	            grant, raw_action, _ = model.select_action(s_vec, deterministic=False)
   494	            open_decision[vid] = {"state": np.asarray(s_vec, dtype=np.float32), "raw_action": raw_action}
   495	            action_dict[vid] = grant
   496	
   497	        for vid in [v for v in open_decision if v not in next_obs]:
   498	            open_decision.pop(vid, None)
   499	
   500	        obs = next_obs
   501	
   502	    # Close the SMDP intervals still in flight when the step budget ran out, and
   503	    # push their transitions like any other completed interval. `close()` used to
   504	    # discard them: 12.3 % of the decisions made in a 600-step episode never
   505	    # reached the buffer, and the loss is length-biased (an interval is likelier
   506	    # to still be open the longer its Delta), so what went missing was
   507	    # disproportionately the long-Delta decisions carrying the largest accrued
   508	    # penalty. The buffer therefore saw long-Delta successes but not long-Delta
   509	    # failures. These records come back with done=False and transmitted=False.
   510	    for rec in env.finalize_open_intervals():
   511	        if buffer is None:
   512	            continue
   513	        prev = open_decision.pop(rec["vid"], None)
   514	        if prev is None:
   515	            continue
   516	        s2 = obs.get(rec["vid"])
   517	        if s2 is None:
   518	            s2 = np.zeros(STATE_DIM, dtype=np.float32)
   519	        buffer.push(prev["state"], prev["raw_action"], rec["reward"],
   520	                    np.asarray(s2, dtype=np.float32), bool(rec["done"]), rec["delta_actual"])
   521	
   522	    metrics = env.get_metrics()
   523	    env.close()
   524	    gc.collect()
   525	
   526	    # Normalise against the ACTUAL decoder power bounds (design range [10, 23] dBm),
   527	    # not a hardcoded [20, 30] window. `normalize_power_dbm` is imported from
   528	    # src/evaluate.py so the HPO objective and the leaderboard that ranks the
   529	    # tuned models cannot use two different power scales; they used to.
   530	    avg_p = metrics.get("avg_tx_power_dbm", 0.5 * (P_MIN + P_MAX))
   531	    metrics["avg_power_dbm"] = avg_p
   532	    metrics["avg_power_norm"] = round(normalize_power_dbm(avg_p), 4)
   533	    metrics["n_update_failures"] = int(n_update_failures)
   534	
   535	    # Outage is COVERAGE outage (user decision, 2026-08-31): the fraction of
   536	    # vehicle-time spent outside the RSU range, where no AoI update can happen.
   537	    # `packet_loss_rate` is the link-layer frame error rate and is a different
   538	    # quantity; this line used to alias one to the other, which is why the two
   539	    # columns in results/hpo/optuna_trials_*.csv were identical. The environment
   540	    # owns the measurement and must export `OUTAGE_METRIC_KEY`.
   541	    if OUTAGE_METRIC_KEY in metrics:
   542	        metrics["outage_rate"] = metrics[OUTAGE_METRIC_KEY]
   543	        metrics["outage_metric"] = OUTAGE_METRIC_KEY
   544	    else:
   545	        metrics["outage_rate"] = metrics.get(LEGACY_OUTAGE_METRIC_KEY, 0.0)
   546	        metrics["outage_metric"] = LEGACY_OUTAGE_METRIC_KEY
   547	        logger.warning(
   548	            "AoiV2IEnv.get_metrics() did not return '%s'; the HPO objective is scoring "
   549	            "'%s' (link-layer frame error rate) instead. These are different quantities "
   550	            "and the tuned hyperparameters inherit the substitution.",
   551	            OUTAGE_METRIC_KEY, LEGACY_OUTAGE_METRIC_KEY,
   552	        )
   553	
   554	    # Emptiness check, immediately before returning. `n_observations == 0` or
   555	    # `tx_attempts == 0` means the rollout measured nothing: no AoI sample was
   556	    # recorded, or the policy never transmitted. Every remaining metric has
   557	    # degraded to a flattering zero, and `compute_composite_objective` would
   558	    # score that 0.0 -- the global minimum of a minimisation study.
   559	    metrics["run_failed"] = bool(run_is_empty(metrics))
   560	    if metrics["run_failed"]:
   561	        logger.warning(
   562	            "Empty rollout (seed=%s, density=%s, n_steps=%s): n_observations=%s, "
   563	            "tx_attempts=%s. Scoring it %.1f instead of 0.0.",
   564	            seed, density, n_steps, metrics.get("n_observations"),
   565	            metrics.get("tx_attempts"), FAILED_RUN_PENALTY,
   566	        )
   567	
   568	    return metrics

   571	def evaluate_trial_multiseed(
   572	    model_cls: Any,
   573	    hparams: Dict[str, Any],
   574	    seeds: List[int],
   575	    n_steps: int = 35,
   576	    density: float = 25.0,
   577	    rsu_range: float = DEFAULT_RSU_RANGE,
   578	    reward_weights: Optional[Dict[str, float]] = None,
   579	    trial: Optional[optuna.Trial] = None,
   580	) -> Tuple[float, Dict[str, float]]:
   581	    """
   582	    Evaluates a set of hyperparameters across multiple seeds and returns the mean composite score.
   583	
   584	    When `trial` is given, every per-seed diagnostic that the CSV needs in order
   585	    to tell a healthy trial from a dead one is written as a user attribute:
   586	    `n_observations_seed<S>`, `tx_attempts_seed<S>`, `n_failed_seeds` (rollouts
   587	    that raised or measured nothing) and `n_update_failures`. Without them a
   588	    study where two of three seeds died looked exactly like a healthy one.
   589	
   590	    `n_vehicles` used to be an argument here and was never read; the population
   591	    is set by `density`.
   592	    """
   593	    scores = []
   594	    seed_metrics: List[Dict[str, float]] = []
   595	    n_failed_seeds = 0
   596	    n_update_failures = 0
   597	
   598	    for i, seed in enumerate(seeds):
   599	        try:
   600	            # Instantiate model
   601	            model = model_cls(state_dim=STATE_DIM, num_channels=comm.NUM_SUBCHANNELS, **hparams)
   602	            metrics = evaluate_model_in_env(
   603	                model=model,
   604	                seed=seed,
   605	                n_steps=n_steps,
   606	                density=density,
   607	                rsu_range=rsu_range,
   608	                train_steps_during_rollout=2,
   609	                reward_weights=reward_weights,
   610	            )
   611	            score = compute_composite_objective(metrics)
   612	            scores.append(score)
   613	            seed_metrics.append(metrics)
   614	            n_update_failures += int(metrics.get("n_update_failures", 0) or 0)
   615	            if metrics.get("run_failed"):
   616	                n_failed_seeds += 1
   617	            if trial is not None:
   618	                trial.set_user_attr(f"n_observations_seed{seed}", int(metrics.get("n_observations", 0) or 0))
   619	                trial.set_user_attr(f"tx_attempts_seed{seed}", int(metrics.get("tx_attempts", 0) or 0))
   620	        except Exception as e:
   621	            logger.exception(f"Error during rollout for seed {seed}: {e}")
   622	            scores.append(FAILED_RUN_PENALTY)
   623	            n_failed_seeds += 1
   624	            if trial is not None:
   625	                trial.set_user_attr(f"n_observations_seed{seed}", -1)
   626	                trial.set_user_attr(f"tx_attempts_seed{seed}", -1)
   627	
   628	        if trial is not None and scores:
   629	            # Report the running mean so a pruner has something to act on. With
   630	            # the default NopPruner this is inert bookkeeping; it exists so that
   631	            # configuring a real pruner actually prunes.
   632	            trial.report(float(np.mean(scores)), step=i)
   633	            if trial.should_prune():
   634	                raise optuna.TrialPruned()
   635	
   636	    mean_score = float(np.mean(scores)) if scores else FAILED_RUN_PENALTY
   637	
   638	    # Average individual metrics across seeds
   639	    avg_metrics: Dict[str, float] = {}
   640	    if seed_metrics:
   641	        for k in ["mean_error", "mean_aoi", "outage_rate", OUTAGE_METRIC_KEY,
   642	                  "packet_loss_rate", "avg_tx_power_dbm", "avg_power_norm",
   643	                  "n_observations", "tx_attempts"]:
   644	            if k in seed_metrics[0]:
   645	                avg_metrics[k] = round(float(np.mean([m.get(k, 0.0) for m in seed_metrics])), 4)
   646	    avg_metrics["n_failed_seeds"] = int(n_failed_seeds)
   647	    avg_metrics["n_update_failures"] = int(n_update_failures)
   648	    if seed_metrics:
   649	        # Which outage definition the objective actually scored, so the trial CSV
   650	        # is unambiguous even if the environment changes underneath it.
   651	        avg_metrics["outage_metric"] = seed_metrics[0].get("outage_metric", "")
   652	
   653	    return mean_score, avg_metrics

   656	def run_hpo_study(
   657	    model_name: Union[str, type, Any],
   658	    model_cls: Optional[Any] = None,
   659	    n_trials: int = 15,
   660	    seeds: Optional[List[int]] = None,
   661	    storage: Optional[str] = None,
   662	    study_name: Optional[str] = None,
   663	    n_steps: int = 350,
   664	    sampler: Optional[optuna.samplers.BaseSampler] = None,
   665	    pruner: Optional[optuna.pruners.BasePruner] = None,
   666	    tune_reward_weights: bool = False,
   667	) -> optuna.Study:
   668	    """
   669	    Runs an Optuna study to optimize hyperparameters for a given model.
   670	
   671	    `tune_reward_weights` is off by default: w1..w4 stay pinned to
   672	    `DEFAULT_REWARD_WEIGHTS` so all nine baselines are searched against one
   673	    identical reward. Turn it on only for a reward-shaping ablation, never for
   674	    the numbers that go into the cross-model comparison table.
   675	    """
   676	    if model_cls is None:
   677	        if callable(model_name) and not isinstance(model_name, str):
   678	            model_cls = model_name
   679	            canonical_name = getattr(model_cls, "__name__", "CustomModel")
   680	        else:
   681	            # Resolve through the baseline registry; it raises with a listing on a miss.
   682	            canonical_name = normalize_model_name(model_name)
   683	            model_cls = get_baseline(canonical_name)
   684	    else:
   685	        canonical_name = normalize_model_name(model_name) if isinstance(model_name, str) else getattr(model_cls, "__name__", "CustomModel")
   686	    eval_seeds = seeds if seeds is not None else list(HPO_TUNING_SEEDS)
   687	
   688	    if sampler is None:
   689	        sampler = optuna.samplers.TPESampler(seed=42)
   690	    if pruner is None:
   691	        # Explicitly no pruning. The previous default was a MedianPruner that
   692	        # could never fire because nothing in the objective called
   693	        # `trial.report()`, so the configuration only looked like pruning.
   694	        # `evaluate_trial_multiseed` now reports the running mean per seed, so
   695	        # passing a real pruner in works; the default stays off because pruning
   696	        # on a 3-seed mean would drop trials on one unlucky seed.
   697	        pruner = optuna.pruners.NopPruner()
   698	
   699	    optuna.logging.set_verbosity(optuna.logging.WARNING)
   700	
   701	    study = optuna.create_study(
   702	        study_name=study_name or f"hpo_{canonical_name}",
   703	        direction="minimize",
   704	        storage=storage,
   705	        sampler=sampler,
   706	        pruner=pruner,
   707	        load_if_exists=True,
   708	    )
   709	
   710	    def objective(trial: optuna.Trial) -> float:
   711	        hparams = sample_hparams(trial, canonical_name)
   712	        # Reward weights are a property of the benchmark, so by default every
   713	        # trial of every model runs under the same fixed reward and only the
   714	        # model hyperparameters are searched. Tuning them per model would give
   715	        # each baseline its own objective and break the cross-model comparison.
   716	        if tune_reward_weights:
   717	            reward_weights = sample_reward_weights(trial)
   718	        else:
   719	            reward_weights = dict(DEFAULT_REWARD_WEIGHTS)
   720	            for k, v in reward_weights.items():
   721	                trial.set_user_attr(k, v)
   722	        score, avg_metrics = evaluate_trial_multiseed(
   723	            model_cls=model_cls,
   724	            hparams=hparams,
   725	            seeds=eval_seeds,
   726	            n_steps=n_steps,
   727	            reward_weights=reward_weights,
   728	            trial=trial,
   729	        )
   730	        for k, v in avg_metrics.items():
   731	            trial.set_user_attr(k, v)
   732	        trial.set_user_attr("composite_score", score)
   733	        return score
   734	
   735	    study.optimize(objective, n_trials=n_trials, n_jobs=1)
   736	    logger.info(
   737	        f"[{canonical_name}] HPO Complete! Best Trial #{study.best_trial.number}: "
   738	        f"Value={study.best_value:.4f}, Params={study.best_params}"
   739	    )
   740	    return study

   743	def save_study_results(
   744	    study: optuna.Study,
   745	    model_name: str,
   746	    output_dir: str = "/home/imnyj/Workspace/paper4/coder/results/hpo",
   747	) -> Tuple[str, Dict[str, Any]]:
   748	    """
   749	    Saves Optuna study trial history to CSV.
   750	    """
   751	    os.makedirs(output_dir, exist_ok=True)
   752	    canonical_name = normalize_model_name(model_name)
   753	
   754	    trials_csv_path = os.path.join(output_dir, f"optuna_trials_{canonical_name}.csv")
   755	    df_trials = study.trials_dataframe()
   756	
   757	    # Guarantee the four *normalised* reward weights appear as plain w1..w4 columns.
   758	    # `trials_dataframe()` only carries the raw suggestions (params_w1_raw ...) and
   759	    # `user_attrs_w1 ...`; the audit found the effective weights were absent entirely.
   760	    attr_by_number = {t.number: t.user_attrs for t in study.trials}
   761	    if "number" in df_trials.columns:
   762	        for key in REWARD_WEIGHT_KEYS:
   763	            df_trials[key] = [
   764	                attr_by_number.get(int(n), {}).get(key) for n in df_trials["number"]
   765	            ]
   766	    df_trials.to_csv(trials_csv_path, index=False)
   767	
   768	    best_params = dict(study.best_params)
   769	    best_weights = {
   770	        k: study.best_trial.user_attrs[k]
   771	        for k in REWARD_WEIGHT_KEYS
   772	        if k in study.best_trial.user_attrs
   773	    }
   774	    # Deliberately NOT merged into best_params: `hparams_json` is consumed by
   775	    # run_all.py and forwarded to a model constructor, and w1..w4 are AoiV2IEnv
   776	    # arguments. They stay in their own `w1`..`w4` / `reward_weights_json`
   777	    # columns, which is where the environment reads them from.
   778	
   779	    best_record = {
   780	        "model_name": canonical_name,
   781	        "category": MODEL_CATEGORIES.get(canonical_name, "Baseline"),
   782	        "best_value": round(float(study.best_value), 4),
   783	        "best_trial_number": int(study.best_trial.number),
   784	        "best_params": best_params,
   785	        "best_reward_weights": best_weights,
   786	    }
   787	    return trials_csv_path, best_record

   790	def run_all_baselines_hpo(
   791	    n_trials: int = 15,
   792	    output_dir: str = "/home/imnyj/Workspace/paper4/coder/results/hpo",
   793	    models: Optional[List[str]] = None,
   794	    seeds: Optional[List[int]] = None,
   795	    n_steps: int = 350,
   796	    tune_reward_weights: bool = False,
   797	) -> Tuple[str, pd.DataFrame]:
   798	    """
   799	    Runs hyperparameter optimization across all baseline models, generates per-model trial CSVs,
   800	    and consolidates the optimal hyperparameters into optuna_best_params.csv.
   801	    """
   802	    os.makedirs(output_dir, exist_ok=True)
   803	    target_models = [normalize_model_name(m) for m in (models or CANONICAL_MODEL_NAMES)]
   804	    eval_seeds = seeds or list(HPO_TUNING_SEEDS)
   805	
   806	    best_records: List[Dict[str, Any]] = []
   807	
   808	    for model_name in target_models:
   809	        logger.info(f"===> Starting Optuna HPO for baseline: {model_name} (trials={n_trials})")
   810	        study = run_hpo_study(
   811	            model_name=model_name,
   812	            n_trials=n_trials,
   813	            seeds=eval_seeds,
   814	            n_steps=n_steps,
   815	            tune_reward_weights=tune_reward_weights,
   816	        )
   817	        _, record = save_study_results(study, model_name=model_name, output_dir=output_dir)
   818	
   819	        # Flatten record for master CSV
   820	        flat_row: Dict[str, Any] = {
   821	            "model_name": record["model_name"],
   822	            "category": record["category"],
   823	            "best_value": record["best_value"],
   824	            "best_trial_number": record["best_trial_number"],
   825	        }
   826	        for k, v in record["best_params"].items():
   827	            flat_row[k] = v
   828	        # Explicit reward-weight columns (Conversation.md line 31 requires w1..w4 to be
   829	        # searched; the audit found them missing from results/hpo/*.csv).
   830	        for k in REWARD_WEIGHT_KEYS:
   831	            flat_row[k] = record.get("best_reward_weights", {}).get(k)
   832	        flat_row["reward_weights_json"] = json.dumps(record.get("best_reward_weights", {}))
   833	        flat_row["hparams_json"] = json.dumps(record["best_params"])
   834	        best_records.append(flat_row)
   835	
   836	    df_best = pd.DataFrame(best_records)
   837	    best_csv_path = os.path.join(output_dir, "optuna_best_params.csv")
   838	    df_best.to_csv(best_csv_path, index=False)
   839	    logger.info(f"All baselines HPO complete! Master parameters saved to {best_csv_path}")
   840	
   841	    return best_csv_path, df_best

   844	def main() -> None:
   845	    parser = argparse.ArgumentParser(description="Optuna HPO Pipeline for AoI Baseline Models")
   846	    parser.add_argument("--n-trials", type=int, default=15, help="Number of trials per model")
   847	    parser.add_argument("--output-dir", type=str, default="/home/imnyj/Workspace/paper4/coder/results/hpo")
   848	    parser.add_argument("--models", nargs="+", default=CANONICAL_MODEL_NAMES, help="List of models to optimize")
   849	    parser.add_argument("--seeds", nargs="+", type=int, default=list(HPO_TUNING_SEEDS),
   850	                        help="Tuning seeds (disjoint from training 42..141 and evaluation 5001..5005)")
   851	    # This is the MEASUREMENT window, not the warmup; the warmup is
   852	    # `DEFAULT_WARMUP_STEPS` and is applied identically here, in training and in
   853	    # evaluation. An earlier comment here read "350 matches the training warmup",
   854	    # which conflated the two: it was never a warmup and it no longer matches one.
   855	    #
   856	    # Why 350 and not 35: 35 steps is 3.5 simulated seconds, and a vehicle needs
   857	    # ~35 s from its spawn edge to reach the RSU disc, so a 35-step window scores
   858	    # almost no traffic and the search optimises noise.
   859	    #
   860	    # Why 350 and not 2000: a training episode and an evaluation episode are both
   861	    # 2000 steps, so the HPO objective is measured over a 5.7x shorter window than
   862	    # the score it is a proxy for. Raising it to 2000 costs ~5.7x the HPO wall
   863	    # clock on top of the warmup increase and invalidates
   864	    # `results/hpo/optuna_best_params.csv`, so it is left at 350 pending an
   865	    # explicit decision rather than changed silently.
   866	    parser.add_argument("--n-steps", type=int, default=350, help="Simulation steps per seed")
   867	    parser.add_argument("--tune-reward-weights", action="store_true",
   868	                        help="search w1..w4 per model instead of pinning them to the "
   869	                             "benchmark defaults (ablation only -- makes cross-model "
   870	                             "comparison invalid)")
   871	    args = parser.parse_args()
   872	
   873	    run_all_baselines_hpo(
   874	        n_trials=args.n_trials,
   875	        output_dir=args.output_dir,
   876	        models=args.models,
   877	        seeds=args.seeds,
   878	        n_steps=args.n_steps,
   879	        tune_reward_weights=args.tune_reward_weights,
   880	    )

   884	    main()
