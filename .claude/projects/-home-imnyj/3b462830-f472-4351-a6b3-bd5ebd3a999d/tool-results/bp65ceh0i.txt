    36	def _write_csv(rows):
    37	    """Write `rows` to a throwaway CSV and return its path."""
    38	    path = os.path.join(tempfile.gettempdir(), f"hparams_{uuid.uuid4().hex}.csv")
    39	    pd.DataFrame(rows).to_csv(path, index=False)
    40	    return path
