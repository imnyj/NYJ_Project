    16	class Building:
    17	    """
    18	    3D Bounding Box model for urban buildings.
    19	    """
    20	    def __init__(
    21	        self,
    22	        building_id: str,
    23	        x_min: float,
    24	        x_max: float,
    25	        y_min: float,
    26	        y_max: float,
    27	        height: float,
    28	        z_min: float = 0.0
    29	    ):
    30	        self.building_id = building_id
    31	        self.x_min = min(x_min, x_max)
    32	        self.x_max = max(x_min, x_max)
    33	        self.y_min = min(y_min, y_max)
    34	        self.y_max = max(y_min, y_max)
    35	        self.z_min = z_min
    36	        self.height = height
    37	        self.z_max = z_min + height
    38	
    39	    @property
    40	    def center(self) -> Tuple[float, float, float]:
    41	        return (
    42	            (self.x_min + self.x_max) / 2.0,
    43	            (self.y_min + self.y_max) / 2.0,
    44	            (self.z_min + self.z_max) / 2.0
    45	        )
    46	
    47	    def contains_point(self, point: Tuple[float, float, float], margin: float = 0.0) -> bool:
    48	        x, y, z = point
    49	        return (
    50	            (self.x_min - margin <= x <= self.x_max + margin) and
    51	            (self.y_min - margin <= y <= self.y_max + margin) and
    52	            (self.z_min - margin <= z <= self.z_max + margin)
    53	        )
    54	
    55	    def intersects_line_2d(
    56	        self,
    57	        p1: Tuple[float, float],
    58	        p2: Tuple[float, float],
    59	        margin: float = 0.0
    60	    ) -> bool:
    61	        """
    62	        Check if 2D line segment p1-p2 intersects building 2D bounding box extended by margin.
    63	        """
    64	        x1, y1 = p1
    65	        x2, y2 = p2
    66	
    67	        min_x = self.x_min - margin
    68	        max_x = self.x_max + margin
    69	        min_y = self.y_min - margin
    70	        max_y = self.y_max + margin
    71	
    72	        # Quick bounding box overlap check
    73	        if max(x1, x2) < min_x or min(x1, x2) > max_x:
    74	            return False
    75	        if max(y1, y2) < min_y or min(y1, y2) > max_y:
    76	            return False
    77	
    78	        # Liang-Barsky line clipping
    79	        dx = x2 - x1
    80	        dy = y2 - y1
    81	
    82	        p = [-dx, dx, -dy, dy]
    83	        q = [x1 - min_x, max_x - x1, y1 - min_y, max_y - y1]
    84	
    85	        u1 = 0.0
    86	        u2 = 1.0
    87	
    88	        for i in range(4):
    89	            if p[i] == 0:
    90	                if q[i] < 0:
    91	                    return False
    92	            else:
    93	                t = q[i] / p[i]
    94	                if p[i] < 0:
    95	                    u1 = max(u1, t)
    96	                else:
    97	                    u2 = min(u2, t)
    98	
    99	        return u1 <= u2
   100	
   101	    def intersects_segment_3d(
   102	        self,
   103	        p1: Tuple[float, float, float],
   104	        p2: Tuple[float, float, float],
   105	        margin: float = 0.0
   106	    ) -> bool:
   107	        """
   108	        Check if 3D line segment p1-p2 intersects 3D building bounding box (with safety margin).
   109	        """
   110	        # First check vertical range
   111	        z_min_seg = min(p1[2], p2[2])
   112	        z_max_seg = max(p1[2], p2[2])
   113	
   114	        b_zmin = self.z_min - margin
   115	        b_zmax = self.z_max + margin
   116	
   117	        if z_max_seg < b_zmin or z_min_seg > b_zmax:
   118	            return False
   119	
   120	        # Check 2D projection intersection
   121	        return self.intersects_line_2d((p1[0], p1[1]), (p2[0], p2[1]), margin=margin)
   122	
   123	    def get_corners_2d(self, margin: float = 0.0) -> List[Tuple[float, float]]:
   124	        """
   125	        Return 4 outer corners in 2D with safety margin.
   126	        """
   127	        return [
   128	            (self.x_min - margin, self.y_min - margin),
   129	            (self.x_max + margin, self.y_min - margin),
   130	            (self.x_max + margin, self.y_max + margin),
   131	            (self.x_min - margin, self.y_max + margin),
   132	        ]

   135	class Vertiport:
   136	    """
   137	    Vertiport model for eVTOL takeoff/landing with time-slot scheduling.
   138	    """
   139	    def __init__(self, vertiport_id: str, position: Tuple[float, float, float], capacity: int = 2):
   140	        self.vertiport_id = vertiport_id
   141	        self.position = np.array(position, dtype=float)
   142	        self.capacity = capacity
   143	        self.reservations: List[Tuple[float, float, str]] = []  # (start_time, end_time, vehicle_id)
   144	
   145	    def is_available(self, start_time: float, duration: float) -> bool:
   146	        end_time = start_time + duration
   147	        overlapping = [
   148	            res for res in self.reservations
   149	            if not (end_time <= res[0] or start_time >= res[1])
   150	        ]
   151	        return len(overlapping) < self.capacity
   152	
   153	    def reserve_slot(self, start_time: float, duration: float, vehicle_id: str) -> bool:
   154	        if self.is_available(start_time, duration):
   155	            self.reservations.append((start_time, start_time + duration, vehicle_id))
   156	            return True
   157	        return False

   164	class RoutePlanner:
   165	    """
   166	    3D Route Planner for eVTOLs avoiding building bounding boxes.
   167	    Generates 3D waypoints including takeoff, cruise detour, and landing stages.
   168	    """
   169	    def __init__(
   170	        self,
   171	        default_cruise_altitude: float = 200.0,
   172	        safety_margin_horizontal: float = 30.0,
   173	        safety_margin_vertical: float = 30.0
   174	    ):
   175	        self.default_cruise_altitude = default_cruise_altitude
   176	        self.safety_margin_h = safety_margin_horizontal
   177	        self.safety_margin_v = safety_margin_vertical
   178	
   179	    def calculate_required_altitude(
   180	        self,
   181	        start_pos: Tuple[float, float, float],
   182	        dest_pos: Tuple[float, float, float],
   183	        buildings: List[Building]
   184	    ) -> float:
   185	        """
   186	        Calculate minimum required cruise altitude along direct flight path to safely clear buildings.
   187	        """
   188	        max_b_height = 0.0
   189	        p1_2d = (start_pos[0], start_pos[1])
   190	        p2_2d = (dest_pos[0], dest_pos[1])
   191	
   192	        for b in buildings:
   193	            if b.intersects_line_2d(p1_2d, p2_2d, margin=self.safety_margin_h):
   194	                if b.z_max > max_b_height:
   195	                    max_b_height = b.z_max
   196	
   197	        req_altitude = max(self.default_cruise_altitude, max_b_height + self.safety_margin_v)
   198	        return req_altitude
   199	
   200	    def plan_3d_route(
   201	        self,
   202	        start_pos: Tuple[float, float, float],
   203	        dest_pos: Tuple[float, float, float],
   204	        buildings: List[Building],
   205	        cruise_altitude: Optional[float] = None
   206	    ) -> List[Tuple[float, float, float]]:
   207	        """
   208	        Generate 3D waypoints:
   209	        1. Vertical takeoff from start_pos to cruise_altitude
   210	        2. Horizontal detour navigation around building bounding boxes
   211	        3. Vertical descent from cruise_altitude to dest_pos
   212	        """
   213	        if cruise_altitude is None:
   214	            cruise_altitude = self.calculate_required_altitude(start_pos, dest_pos, buildings)
   215	
   216	        start_x, start_y, start_z = start_pos
   217	        dest_x, dest_y, dest_z = dest_pos
   218	
   219	        waypoints = []
   220	
   221	        # 1. Start point
   222	        waypoints.append((start_x, start_y, start_z))
   223	
   224	        # 2. Takeoff waypoint (vertical climb to cruise altitude)
   225	        waypoints.append((start_x, start_y, cruise_altitude))
   226	
   227	        # 3. Horizontal segment planning at cruise altitude
   228	        current_2d = (start_x, start_y)
   229	        target_2d = (dest_x, dest_y)
   230	
   231	        # Check if buildings collide with flight path at cruise altitude
   232	        colliding_buildings = [
   233	            b for b in buildings
   234	            if b.z_max + self.safety_margin_v > cruise_altitude and
   235	               b.intersects_line_2d(current_2d, target_2d, margin=self.safety_margin_h)
   236	        ]
   237	
   238	        if colliding_buildings:
   239	            # Generate detour waypoints around colliding buildings
   240	            detour_2d = self._generate_detours_2d(current_2d, target_2d, colliding_buildings)
   241	            for dx, dy in detour_2d:
   242	                waypoints.append((dx, dy, cruise_altitude))
   243	        else:
   244	            # Direct cruise waypoint before descent
   245	            waypoints.append((dest_x, dest_y, cruise_altitude))
   246	
   247	        # 4. Final landing waypoint (vertical descent to destination)
   248	        waypoints.append((dest_x, dest_y, dest_z))
   249	
   250	        return waypoints
   251	
   252	    def _generate_detours_2d(
   253	        self,
   254	        start_2d: Tuple[float, float],
   255	        dest_2d: Tuple[float, float],
   256	        colliding_buildings: List[Building]
   257	    ) -> List[Tuple[float, float]]:
   258	        """
   259	        Simple geometric detour generation around building bounding box corners.
   260	        """
   261	        waypoints_2d = []
   262	        curr = start_2d
   263	
   264	        for b in colliding_buildings:
   265	            if not b.intersects_line_2d(curr, dest_2d, margin=self.safety_margin_h):
   266	                continue
   267	
   268	            corners = b.get_corners_2d(margin=self.safety_margin_h)
   269	            b_center = (b.center[0], b.center[1])
   270	            
   271	            v_line = (dest_2d[0] - curr[0], dest_2d[1] - curr[1])
   272	            v_b = (b_center[0] - curr[0], b_center[1] - curr[1])
   273	            cross_prod = v_line[0] * v_b[1] - v_line[1] * v_b[0]
   274	
   275	            selected_corners = []
   276	            for cx, cy in corners:
   277	                v_c = (cx - curr[0], cy - curr[1])
   278	                cp = v_line[0] * v_c[1] - v_line[1] * v_c[0]
   279	                if cross_prod >= 0 and cp > 0:
   280	                    selected_corners.append((cx, cy))
   281	                elif cross_prod < 0 and cp < 0:
   282	                    selected_corners.append((cx, cy))
   283	
   284	            if not selected_corners:
   285	                selected_corners = corners
   286	
   287	            best_corner = min(
   288	                selected_corners,
   289	                key=lambda c: math.hypot(c[0] - curr[0], c[1] - curr[1]) + math.hypot(dest_2d[0] - c[0], dest_2d[1] - c[1])
   290	            )
   291	
   292	            waypoints_2d.append(best_corner)
   293	            curr = best_corner
   294	
   295	        waypoints_2d.append(dest_2d)
   296	        return waypoints_2d

   303	class VehicleState(Enum):
   304	    IDLE = "IDLE"
   305	    TAKEOFF = "TAKEOFF"
   306	    CLIMB = "CLIMB"
   307	    CRUISE = "CRUISE"
   308	    DESCENT = "DESCENT"

   311	
   312	
   313	class EVTOLVehicle:
   314	    """
   315	    eVTOL Vehicle class with realistic horizontal/vertical speed constraints and kinematic updates.
   316	    """
   317	    def __init__(
   318	        self,
   319	        vehicle_id: str,
   320	        max_horizontal_speed: Optional[float] = None,
   321	        max_climb_rate: Optional[float] = None,
   322	        max_descent_rate: Optional[float] = None,
   323	        acceleration: Optional[float] = None,
   324	        waypoint_arrival_radius: Optional[float] = None
   325	    ):
   326	        self.vehicle_id = vehicle_id
   327	        cfg = load_config().get("VEHICLE", {})
   328	        self.max_h_speed = max_horizontal_speed if max_horizontal_speed is not None else cfg.get("MAX_HORIZONTAL_SPEED", 50.0)
   329	        self.max_climb_rate = max_climb_rate if max_climb_rate is not None else cfg.get("MAX_CLIMB_RATE", 5.0)
   330	        self.max_descent_rate = max_descent_rate if max_descent_rate is not None else cfg.get("MAX_DESCENT_RATE", 3.0)
   331	        self.acceleration = acceleration if acceleration is not None else cfg.get("ACCELERATION", 2.5)
   332	        self.arrival_radius = waypoint_arrival_radius if waypoint_arrival_radius is not None else cfg.get("ARRIVAL_RADIUS", 5.0)
   333	
   334	        self.position = np.array([0.0, 0.0, 0.0], dtype=float)
   335	        self.velocity = np.array([0.0, 0.0, 0.0], dtype=float)
   336	        self.speed = 0.0
   337	        self.heading_rad = 0.0
   338	
   339	        self.waypoints: List[np.ndarray] = []
   340	        self.current_wp_idx: int = 0
   341	        self.state: VehicleState = VehicleState.IDLE
   342	
   343	        self.connected_node_id: Optional[str] = None
   344	        self.connected_comm_type: Optional[str] = None
   345	        self.handover_history: List[Dict] = []
   346	        self.telemetry_history: List[Dict] = []
   347	        self.total_flight_time: float = 0.0
   348	        self.total_flight_distance: float = 0.0
   349	
   350	    def set_route(self, waypoints: List[Tuple[float, float, float]], start_time: float = 0.0):
   351	        self.waypoints = [np.array(wp, dtype=float) for wp in waypoints]
   352	        if self.waypoints:
   353	            self.position = self.waypoints[0].copy()
   354	            self.current_wp_idx = 1 if len(self.waypoints) > 1 else 0
   355	            self.state = VehicleState.TAKEOFF
   356	        self.total_flight_time = start_time
   357	
   358	    def update(self, dt: float) -> VehicleState:
   359	        """
   360	        Kinematic position update moving vehicle towards current target waypoint.
   361	        Enforces distinct horizontal and vertical speed limits.
   362	        """
   363	        if self.state in (VehicleState.IDLE, VehicleState.ARRIVED) or not self.waypoints:
   364	            return self.state
   365	
   366	        if self.current_wp_idx >= len(self.waypoints):
   367	            self.state = VehicleState.ARRIVED
   368	            self.velocity = np.array([0.0, 0.0, 0.0])
   369	            return self.state
   370	
   371	        self.total_flight_time += dt
   372	        target_wp = self.waypoints[self.current_wp_idx]
   373	        delta = target_wp - self.position
   374	        dist_3d = np.linalg.norm(delta)
   375	
   376	        # Check waypoint arrival
   377	        if dist_3d <= self.arrival_radius:
   378	            self.current_wp_idx += 1
   379	            if self.current_wp_idx >= len(self.waypoints):
   380	                self.state = VehicleState.ARRIVED
   381	                self.velocity = np.array([0.0, 0.0, 0.0])
   382	                return self.state
   383	            target_wp = self.waypoints[self.current_wp_idx]
   384	            delta = target_wp - self.position
   385	            dist_3d = np.linalg.norm(delta)
   386	
   387	        delta_xy = delta[:2]
   388	        dist_xy = np.linalg.norm(delta_xy)
   389	        delta_z = delta[2]
   390	
   391	        # Determine flight phase/state and velocity bounds
   392	        if abs(delta_z) > 1.0 and dist_xy < 10.0:
   393	            if delta_z > 0:
   394	                self.state = VehicleState.CLIMB
   395	                target_vz = min(self.max_climb_rate, delta_z / dt)
   396	            else:
   397	                self.state = VehicleState.DESCENT
   398	                target_vz = max(-self.max_descent_rate, delta_z / dt)
   399	            target_vx = 0.0
   400	            target_vy = 0.0
   401	        else:
   402	            self.state = VehicleState.CRUISE
   403	            if dist_xy > 0:
   404	                dir_xy = delta_xy / dist_xy
   405	                target_h_speed = min(self.max_h_speed, dist_xy / dt)
   406	                target_vx = dir_xy[0] * target_h_speed
   407	                target_vy = dir_xy[1] * target_h_speed
   408	                self.heading_rad = math.atan2(dir_xy[1], dir_xy[0])
   409	            else:
   410	                target_vx, target_vy = 0.0, 0.0
   411	
   412	            if delta_z > 0:
   413	                target_vz = min(self.max_climb_rate, delta_z / dt)
   414	            elif delta_z < 0:
   415	                target_vz = max(-self.max_descent_rate, delta_z / dt)
   416	            else:
   417	                target_vz = 0.0
   418	
   419	        target_velocity = np.array([target_vx, target_vy, target_vz], dtype=float)
   420	
   421	        # Smooth velocity acceleration update
   422	        vel_diff = target_velocity - self.velocity
   423	        vel_diff_norm = np.linalg.norm(vel_diff)
   424	        if vel_diff_norm > 0:
   425	            acc_step = min(self.acceleration * dt, vel_diff_norm)
   426	            self.velocity += (vel_diff / vel_diff_norm) * acc_step
   427	
   428	        # Update position
   429	        move_step = self.velocity * dt
   430	        self.position += move_step
   431	        step_dist = np.linalg.norm(move_step)
   432	        self.total_flight_distance += step_dist
   433	        self.speed = np.linalg.norm(self.velocity)
   434	
   435	        # Record telemetry
   436	        self.telemetry_history.append({
   437	            "timestamp": self.total_flight_time,
   438	            "x": self.position[0],
   439	            "y": self.position[1],
   440	            "z": self.position[2],
   441	            "speed": self.speed,
   442	            "state": self.state.value,
   443	            "comm_node": self.connected_node_id,
   444	            "comm_type": self.connected_comm_type
   445	        })
   446	
   447	        return self.state
   448	
   449	    def get_status(self) -> Dict:
   450	        return {
   451	            "vehicle_id": self.vehicle_id,
   452	            "position": self.position.tolist(),
   453	            "velocity": self.velocity.tolist(),
   454	            "speed": float(self.speed),
   455	            "altitude": float(self.position[2]),
   456	            "state": self.state.value,
   457	            "connected_node": self.connected_node_id,
   458	            "comm_type": self.connected_comm_type,
   459	            "flight_time": self.total_flight_time,
   460	            "flight_distance": self.total_flight_distance
   461	        }

   468	class CommType(Enum):
   469	    CELLULAR_5G = "5G"
   470	    RSU = "RSU"

   472	
   473	
   474	class BaseCommNode:
   475	    """
   476	    Base class for communication nodes in the UAM environment.
   477	    """
   478	    def __init__(
   479	        self,
   480	        node_id: str,
   481	        node_type: CommType,
   482	        position: Tuple[float, float, float],
   483	        frequency_ghz: float,
   484	        tx_power_dbm: float,
   485	        max_range: float,
   486	        bandwidth_mhz: float,
   487	        base_data_rate_mbps: float,
   488	        base_latency_ms: float,
   489	        max_altitude: float = 1000.0
   490	    ):
   491	        self.node_id = node_id
   492	        self.node_type = node_type
   493	        self.position = np.array(position, dtype=float)
   494	        self.frequency_ghz = frequency_ghz
   495	        self.tx_power_dbm = tx_power_dbm
   496	        self.max_range = max_range
   497	        self.bandwidth_mhz = bandwidth_mhz
   498	        self.base_data_rate_mbps = base_data_rate_mbps
   499	        self.base_latency_ms = base_latency_ms
   500	        self.max_altitude = max_altitude
   501	
   502	    def calculate_distance_3d(self, target_pos: Tuple[float, float, float]) -> float:
   503	        return float(np.linalg.norm(self.position - np.array(target_pos)))
   504	
   505	    def calculate_path_loss(self, distance_m: float) -> float:
   506	        """
   507	        Free Space Path Loss (FSPL) model in dB.
   508	        PL(dB) = 20 log10(d_km) + 20 log10(f_GHz) + 92.45
   509	        """
   510	        d_km = max(distance_m / 1000.0, 0.001)
   511	        pl_db = 20.0 * math.log10(d_km) + 20.0 * math.log10(self.frequency_ghz) + 92.45
   512	        return pl_db
   513	
   514	    def calculate_snr(self, distance_m: float, is_los: bool = True) -> float:
   515	        """
   516	        Calculate Signal-to-Noise Ratio (SNR) in dB.
   517	        """
   518	        pl_db = self.calculate_path_loss(distance_m)
   519	        if not is_los:
   520	            pl_db += 20.0  # NLOS building attenuation penalty
   521	
   522	        bw_hz = self.bandwidth_mhz * 1e6
   523	        noise_floor_dbm = -174.0 + 10.0 * math.log10(bw_hz) + 5.0  # 5dB Noise figure
   524	        rx_power_dbm = self.tx_power_dbm - pl_db
   525	        snr_db = rx_power_dbm - noise_floor_dbm
   526	        return snr_db
   527	
   528	    def calculate_link_quality(
   529	        self,
   530	        target_pos: Tuple[float, float, float],
   531	        is_los: bool = True
   532	    ) -> Dict[str, float]:
   533	        """
   534	        Calculate link parameters: distance, SNR, data rate (Shannon limit), and latency.
   535	        """
   536	        dist = self.calculate_distance_3d(target_pos)
   537	        target_alt = target_pos[2]
   538	
   539	        if dist > self.max_range or target_alt > self.max_altitude:
   540	            return {
   541	                "distance": dist,
   542	                "snr_db": -999.0,
   543	                "data_rate_mbps": 0.0,
   544	                "latency_ms": 999.0,
   545	                "in_range": False
   546	            }
   547	
   548	        snr_db = self.calculate_snr(dist, is_los=is_los)
   549	        snr_linear = max(10.0 ** (snr_db / 10.0), 1e-6)
   550	
   551	        shannon_rate_mbps = (self.bandwidth_mhz) * math.log2(1.0 + snr_linear)
   552	        achievable_rate = min(self.base_data_rate_mbps, shannon_rate_mbps)
   553	
   554	        prop_delay_ms = (dist / 3e8) * 1000.0
   555	        total_latency = self.base_latency_ms + prop_delay_ms
   556	        if not is_los:
   557	            total_latency += 10.0
   558	
   559	        return {
   560	            "distance": dist,
   561	            "snr_db": snr_db,
   562	            "data_rate_mbps": max(0.0, float(achievable_rate)),
   563	            "latency_ms": float(total_latency),
   564	            "in_range": True
   565	        }

   661	    def calculate_distance_3d(self, target_pos: Tuple[float, float, float]) -> float:
   662	        horizontal_dist = math.hypot(target_pos[0] - self.position[0], target_pos[1] - self.position[1])
   663	        altitude_diff = self.position[2] - target_pos[2]
   664	        return math.hypot(horizontal_dist, altitude_diff)

   667	# ==========================================
   668	# 5. Communication Environment Manager
   669	# ==========================================
   670	
   671	class CommunicationEnvironment:
   672	    """
   673	    Manages all wireless communication infrastructure (5G BSs, RSUs, Starlink)
   674	    and handles link evaluation, line-of-sight checks, and proactive handover triggering.
   675	    """
   676	    def __init__(self, buildings: Optional[List[Building]] = None):
   677	        self.nodes: Dict[str, BaseCommNode] = {}
   678	        self.buildings: List[Building] = buildings if buildings else []
   679	
   680	    def add_node(self, node: BaseCommNode):
   681	        self.nodes[node.node_id] = node
   682	
   683	    def check_line_of_sight(
   684	        self,
   685	        pos1: Tuple[float, float, float],
   686	        pos2: Tuple[float, float, float]
   687	    ) -> bool:
   688	        """
   689	        Check if line between pos1 and pos2 is clear of building obstructions.
   690	        """
   691	        for b in self.buildings:
   692	            if b.intersects_segment_3d(pos1, pos2):
   693	                return False
   694	        return True
   695	
   696	    def get_available_links(
   697	        self,
   698	        vehicle_pos: Tuple[float, float, float]
   699	    ) -> List[Dict]:
   700	        """
   701	        Returns list of all in-range communication nodes and their link metrics.
   702	        """
   703	        available = []
   704	        for node_id, node in self.nodes.items():
   705	            is_los = self.check_line_of_sight(vehicle_pos, tuple(node.position))
   706	            metrics = node.calculate_link_quality(vehicle_pos, is_los=is_los)
   707	            if metrics["in_range"] and metrics["snr_db"] > -10.0:
   708	                available.append({
   709	                    "node_id": node_id,
   710	                    "node_type": node.node_type.value,
   711	                    "distance": metrics["distance"],
   712	                    "snr_db": metrics["snr_db"],
   713	                    "data_rate_mbps": metrics["data_rate_mbps"],
   714	                    "latency_ms": metrics["latency_ms"],
   715	                    "is_los": is_los
   716	                })
   717	        return available
   718	
   719	    def find_best_connection(
   720	        self,
   721	        vehicle_pos: Tuple[float, float, float],
   722	        preference_weights: Optional[Dict[str, float]] = None
   723	    ) -> Optional[Dict]:
   724	        """
   725	        Select best network connection based on weighted data rate, latency, and SNR.
   726	        """
   727	        links = self.get_available_links(vehicle_pos)
   728	        if not links:
   729	            return None
   730	
   731	        if preference_weights is None:
   732	            preference_weights = {"data_rate": 0.5, "latency": 0.3, "snr": 0.2}
   733	
   734	        best_score = -float('inf')
   735	        best_link = None
   736	
   737	        for link in links:
   738	            rate_score = min(link["data_rate_mbps"] / 500.0, 1.0)
   739	            lat_score = max(0.0, 1.0 - (link["latency_ms"] / 100.0))
   740	            snr_score = min(max((link["snr_db"] + 10.0) / 40.0, 0.0), 1.0)
   741	
   742	            score = (
   743	                preference_weights["data_rate"] * rate_score +
   744	                preference_weights["latency"] * lat_score +
   745	                preference_weights["snr"] * snr_score
   746	            )
   747	
   748	            if score > best_score:
   749	                best_score = score
   750	                best_link = link
   751	
   752	        return best_link
   753	
   754	    def evaluate_handover(
   755	        self,
   756	        vehicle: EVTOLVehicle,
   757	        current_time: float,
   758	        handover_hysteresis_db: float = 3.0
   759	    ) -> Optional[Dict]:
   760	        """
   761	        Evaluates whether vehicle should initiate a handover.
   762	        Returns handover record if a handover occurs.
   763	        """
   764	        pos = tuple(vehicle.position)
   765	        best_link = self.find_best_connection(pos)
   766	
   767	        if not best_link:
   768	            return None
   769	
   770	        current_node_id = vehicle.connected_node_id
   771	
   772	        if current_node_id is None:
   773	            vehicle.connected_node_id = best_link["node_id"]
   774	            vehicle.connected_comm_type = best_link["node_type"]
   775	            record = {
   776	                "timestamp": current_time,
   777	                "event": "INITIAL_CONNECT",
   778	                "target_node": best_link["node_id"],
   779	                "comm_type": best_link["node_type"],
   780	                "metrics": best_link
   781	            }
   782	            vehicle.handover_history.append(record)
   783	            return record
   784	
   785	        if current_node_id != best_link["node_id"]:
   786	            curr_node = self.nodes.get(current_node_id)
   787	            if curr_node:
   788	                is_los = self.check_line_of_sight(pos, tuple(curr_node.position))
   789	                curr_metrics = curr_node.calculate_link_quality(pos, is_los=is_los)
   790	
   791	                if not curr_metrics["in_range"] or (best_link["snr_db"] > curr_metrics["snr_db"] + handover_hysteresis_db):
   792	                    old_node = current_node_id
   793	                    vehicle.connected_node_id = best_link["node_id"]
   794	                    vehicle.connected_comm_type = best_link["node_type"]
   795	
   796	                    record = {
   797	                        "timestamp": current_time,
   798	                        "event": "HANDOVER",
   799	                        "from_node": old_node,
   800	                        "to_node": best_link["node_id"],
   801	                        "comm_type": best_link["node_type"],
   802	                        "metrics": best_link
   803	                    }
   804	                    vehicle.handover_history.append(record)
   805	                    return record
   806	
   807	        return None

   810	# ==========================================
   811	# 6. Grid Environment Simulation Helper
   812	# ==========================================
   813	
   814	def generate_grid_environment(
   815	    num_grid_cells_x: int = 3,
   816	    num_grid_cells_y: int = 3,
   817	    cell_size: float = 3000.0,       # 3km between intersections
   818	    border_size: float = 1500.0,     # 1.5km border road
   819	    num_buildings: int = 20,
   820	    seed: int = 42
   821	) -> Tuple[List[Building], CommunicationEnvironment]:
   822	    """
   823	    Generates grid simulation environment based on sketch.md specs:
   824	    - 3km distance between intersections with 1.5km outer roads
   825	    - RSU at every intersection
   826	    - 5G Base Stations across sectors
   827	    - Starlink LEO Satellite continuous coverage
   828	    - Non-overlapping building bounding boxes
   829	    """
   830	    np.random.seed(seed)
   831	    buildings = []
   832	    comm_env = CommunicationEnvironment(buildings=buildings)
   833	
   834	    intersection_coords = []
   835	    rsu_idx = 0
   836	
   837	    total_width = border_size * 2 + num_grid_cells_x * cell_size
   838	    total_height = border_size * 2 + num_grid_cells_y * cell_size
   839	
   840	    for ix in range(num_grid_cells_x + 1):
   841	        x = border_size + ix * cell_size
   842	        for iy in range(num_grid_cells_y + 1):
   843	            y = border_size + iy * cell_size
   844	            intersection_coords.append((x, y))
   845	
   846	            rsu = RSU(
   847	                node_id=f"RSU_{rsu_idx}",
   848	                position=(x, y, 10.0)
   849	            )
   850	            comm_env.add_node(rsu)
   851	            rsu_idx += 1
   852	
   853	    bs_idx = 0
   854	    gnode_spacing = cell_size / 2.0
   855	    for bx in np.arange(border_size / 2, total_width, gnode_spacing):
   856	        for by in np.arange(border_size / 2, total_height, gnode_spacing):
   857	            gnode = Cellular5G(
   858	                node_id=f"5G_BS_{bs_idx}",
   859	                position=(float(bx), float(by), 30.0)
   860	            )
   861	            comm_env.add_node(gnode)
   862	            bs_idx += 1
   863	
   864	    starlink = Starlink(
   865	        node_id="Starlink_LEO_1",
   866	        position=(total_width / 2.0, total_height / 2.0, 550000.0)
   867	    )
   868	    comm_env.add_node(starlink)
   869	
   870	    road_width = 40.0
   871	    attempts = 0
   872	    b_id = 0
   873	
   874	    while len(buildings) < num_buildings and attempts < 1000:
   875	        attempts += 1
   876	        bw = np.random.uniform(50.0, 150.0)
   877	        bl = np.random.uniform(50.0, 150.0)
   878	        bh = np.random.uniform(30.0, 180.0)
   879	
   880	        bx = np.random.uniform(border_size, total_width - border_size - bw)
   881	        by = np.random.uniform(border_size, total_height - border_size - bl)
   882	
   883	        new_b = Building(
   884	            building_id=f"Building_{b_id}",
   885	            x_min=bx,
   886	            x_max=bx + bw,
   887	            y_min=by,
   888	            y_max=by + bl,
   889	            height=bh
   890	        )
   891	
   892	        near_road = False
   893	        for ix_x, ix_y in intersection_coords:
   894	            if abs(new_b.center[0] - ix_x) < (bw / 2 + road_width) and abs(new_b.center[1] - ix_y) < (bl / 2 + road_width):
   895	                near_road = True
   896	                break
   897	
   898	        if near_road:
   899	            continue
   900	
   901	        overlap = False
   902	        for existing in buildings:
   903	            if not (
   904	                new_b.x_max + 10.0 < existing.x_min or
   905	                new_b.x_min - 10.0 > existing.x_max or
   906	                new_b.y_max + 10.0 < existing.y_min or
   907	                new_b.y_min - 10.0 > existing.y_max
   908	            ):
   909	                overlap = True
   910	                break
   911	
   912	        if not overlap:
   913	            buildings.append(new_b)
   914	            b_id += 1
   915	
   916	    return buildings, comm_env
