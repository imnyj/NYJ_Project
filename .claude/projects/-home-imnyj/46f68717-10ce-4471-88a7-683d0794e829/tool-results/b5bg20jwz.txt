    48	def get_sumo_max_red_phase_duration(
    49	    net_file: Optional[str] = None,
    50	    default_duration: float = 45.0,
    51	) -> float:
    52	    """
    53	    Dynamically extract the maximum Red traffic light phase duration (seconds)
    54	    from SUMO's generated network XML file (generated.net.xml) or TraCI.
    55	
    56	    Parses all <tlLogic> program definitions in the net file, computes for each
    57	    signal link across cyclic phases the maximum consecutive duration of 'r'/'R'
    58	    phases (handling cycle wrap-around), and returns the overall maximum.
    59	
    60	    If the network file is not yet generated or cannot be parsed, falls back
    61	    safely to `default_duration` (45.0 s).
    62	    """
    63	    if net_file is None:
    64	        base_dir = os.path.dirname(os.path.abspath(__file__))
    65	        net_file = os.path.join(base_dir, "sumo", "generated.net.xml")
    66	
    67	    if not os.path.exists(net_file):
    68	        return default_duration
    69	
    70	    try:
    71	        tree = ET.parse(net_file)
    72	        root = tree.getroot()
    73	        max_red = 0.0
    74	
    75	        for tl in root.findall("tlLogic"):
    76	            phases = tl.findall("phase")
    77	            if not phases:
    78	                continue
    79	            durations = [float(p.get("duration", 0.0)) for p in phases]
    80	            states = [p.get("state", "") for p in phases]
    81	            if not states or not durations:
    82	                continue
    83	            num_links = max(len(s) for s in states)
    84	            n_phases = len(phases)
    85	
    86	            for link_idx in range(num_links):
    87	                curr_red = 0.0
    88	                max_link_red = 0.0
    89	                # Double the cycle to handle cyclic wrap-around
    90	                for step in range(2 * n_phases):
    91	                    p_idx = step % n_phases
    92	                    char = states[p_idx][link_idx] if link_idx < len(states[p_idx]) else "g"
    93	                    if char in ("r", "R"):
    94	                        curr_red += durations[p_idx]
    95	                        if curr_red > max_link_red:
    96	                            max_link_red = curr_red
    97	                    else:
    98	                        curr_red = 0.0
    99	                total_cycle = sum(durations)
   100	                max_link_red = min(max_link_red, total_cycle)
   101	                if max_link_red > max_red:
   102	                    max_red = max_link_red
   103	
   104	        return float(max_red) if max_red > 0.0 else default_duration
   105	    except Exception:
   106	        return default_duration

   113	def get_sumo_max_edge_speed(
   114	    net_file: Optional[str] = None,
   115	    default_speed: float = 13.32,
   116	) -> float:
   117	    """
   118	    Maximum lane speed limit (m/s) declared anywhere in the generated network.
   119	
   120	    Read from the net file for the same reason `DELTA_MAX` is: a normalisation
   121	    constant that is silently inconsistent with the scenario it normalises is a
   122	    defect waiting to happen. `make_sumo_set.py` randomises each edge's speed
   123	    around `AV_SPEED` with +/- `DEL_SPEED`, so the effective limit is a property
   124	    of the generated network, not a literal anyone can restate correctly.
   125	
   126	    Falls back to `default_speed` (the 40 km/h + 20 % case) when the network has
   127	    not been generated yet.
   128	    """
   129	    if net_file is None:
   130	        base_dir = os.path.dirname(os.path.abspath(__file__))
   131	        net_file = os.path.join(base_dir, "sumo", "generated.net.xml")
   132	
   133	    if not os.path.exists(net_file):
   134	        return default_speed
   135	
   136	    try:
   137	        tree = ET.parse(net_file)
   138	        root = tree.getroot()
   139	        speeds = []
   140	        for edge in root.findall("edge"):
   141	            # Internal junction edges inherit their speed from the corner limit
   142	            # and would drag the maximum down; they are not where vehicles cruise.
   143	            if edge.get("function") == "internal":
   144	                continue
   145	            for lane in edge.findall("lane"):
   146	                v = lane.get("speed")
   147	                if v is not None:
   148	                    speeds.append(float(v))
   149	        return float(max(speeds)) if speeds else default_speed
   150	    except Exception:
   151	        return default_speed

   154	#: Scenario speed limit (m/s), derived from the generated network.
   155	V_LIMIT: float = get_sumo_max_edge_speed()
   156	
   157	#: Estimation-error reference scale (metres), design_spec_v2 D5.
   158	#:
   159	#: One second of travel at the scenario speed limit. The point of dividing a
   160	#: position error by a speed is that it converts metres into an *equivalent age*:
   161	#: `e = E_REF` means the RSU's belief about this vehicle is as wrong as if it had
   162	#: simply not heard from it for one second. That is the natural unit for an AoI
   163	#: paper, and unlike a bare literal it follows the scenario automatically -- raise
   164	#: `AV_SPEED` in make_sumo_set.py and this tracks it.
   165	#:
   166	#: NOT the RSU communication range: `e` is a positioning error, the range is a
   167	#: link-budget quantity, and normalising one by the other pushes the error term
   168	#: roughly two orders of magnitude below the power term. Since `hpo.py` normalises
   169	#: w1..w4 to sum to 1, no weight Optuna can sample recovers from that.
   170	E_REF: float = V_LIMIT * 1.0
   171	
   172	
   173	def norm_sq_error(err_m: float, e_ref: float = 0.0) -> float:
   174	    """Normalise a squared position error into [0, 1) -- design_spec_v2 D5.
   175	
   176	        Norm(e^2) = e^2 / (e^2 + e_ref^2)
   177	
   178	    Chosen over `min(1, e^2 / e_max^2)` because the clipped form has zero
   179	    gradient everywhere past `e_max`. With Delta reaching 45 s a moving vehicle
   180	    leaves that region within about a second, so the clipped form would stop
   181	    distinguishing "slightly stale" from "hopelessly stale" exactly where the
   182	    scheduling decision matters. This form is strictly monotone in `e`, equals
   183	    0.5 at `e = e_ref`, and never saturates.
   184	    """
   185	    ref = float(e_ref) if e_ref and e_ref > 0.0 else E_REF
   186	    e2 = float(err_m) ** 2
   187	    return float(e2 / (e2 + ref * ref)) if (e2 + ref * ref) > 0.0 else 0.0

   190	class StateVectorizer:
   191	    """
   192	    18-dimensional normalized State Vectorizer.
   193	
   194	    Transforms raw vehicle kinematics, RSU spatial metrics, TraCI TLS signal states,
   195	    and channel congestion indicators into a normalized feature vector in [-1.0, 1.0].
   196	
   197	    Features:
   198	    [0]  Last prediction error, normalized: norm_sq_error(e_last) in [0.0, 1.0).
   199	         How wrong the RSU's dead reckoning was at this vehicle's most recent
   200	         update. Replaces the former normalized-age slot, which is identically
   201	         zero at every SMDP decision epoch and therefore carried no signal.
   202	    [1]  Normalized Vx: clip(vx / v_max, -1.0, 1.0)
   203	    [2]  Normalized Vy: clip(vy / v_max, -1.0, 1.0)
   204	    [3]  Normalized Speed: clip(speed / v_max, 0.0, 1.0)
   205	    [4]  Normalized Acceleration: clip(accel / a_max, -1.0, 1.0)
   206	    [5]  Relative X: clip(dx / rsu_range, -1.0, 1.0)
   207	    [6]  Relative Y: clip(dy / rsu_range, -1.0, 1.0)
   208	    [7]  Normalized Distance: clip(dist / rsu_range, 0.0, 1.0)
   209	    [8]  TLS Red one-hot: 1.0 if red else 0.0
   210	    [9]  TLS Yellow one-hot: 1.0 if yellow else 0.0
   211	    [10] TLS Green one-hot: 1.0 if green else 0.0
   212	    [11] Phase remaining time: clip(time_to_switch / 60.0, 0.0, 1.0)
   213	    [12] Distance to stopline: clip(dist_to_stopline / rsu_range, 0.0, 1.0)
   214	    [13] Active vehicles in cell: clip(n_active / 100.0, 0.0, 1.0)
   215	    [14] Channel Busy Ratio (CBR): clip(cbr, 0.0, 1.0)
   216	    [15] n_queue (Conversation.md S1): normalized number of vehicles queued ahead in the
   217	         same lane -- clip(n_queue / queue_max, 0.0, 1.0)
   218	    [16] heading (Conversation.md S1): signed approach/recede indicator w.r.t. the RSU --
   219	         cos(angle between velocity vector and the vehicle->RSU vector) in [-1.0, 1.0].
   220	         +1.0 = driving straight at the RSU, -1.0 = driving straight away, 0.0 = stopped
   221	         or moving tangentially.
   222	    """
   223	
   224	    #: Canonical observation dimension (module-level single source of truth).
   225	    STATE_DIM: int = STATE_DIM
   226	
   227	    def __init__(
   228	        self,
   229	        rsu_range: float = 300.0,
   230	        v_max: float = 0.0,
   231	        a_max: float = 5.0,
   232	        queue_max: float = 20.0,
   233	    ) -> None:
   234	        self.rsu_range = float(rsu_range)
   235	        # 0 means "derive from the scenario". The former literal 30.0 m/s was
   236	        # 2.3x the fastest lane in the generated network, so the speed features
   237	        # only ever used the bottom 44 % of their range.
   238	        self.v_max = float(v_max) if v_max and v_max > 0.0 else V_LIMIT
   239	        self.a_max = float(a_max)
   240	        self.queue_max = max(1.0, float(queue_max))
   241	
   242	    @property
   243	    def state_dim(self) -> int:
   244	        """Dimension of the emitted observation vector. Read this instead of hardcoding."""
   245	        return STATE_DIM

   247	    # ------------------------------------------------------------------
   248	    # Feature helpers for the two design-mandated features (n_queue, heading)
   249	    # ------------------------------------------------------------------
   250	    @staticmethod
   251	    def _extract_queue_count(*sources: Optional[Dict[str, Any]]) -> float:
   252	        """
   253	        Recover the number of vehicles queued ahead in the same lane from whatever
   254	        telemetry the dynamics/TLS feature dicts expose.
   255	
   256	        Preferred keys are a real halting count published by the SUMO layer
   257	        (TraCI lane.getLastStepHaltingNumber). If none is present, fall back to the
   258	        leader-vehicle signal already produced by dynamics_predictor.extract_tls_features:
   259	        a stopped/crawling leader within a plausible queue gap implies at least one
   260	        vehicle queued ahead.
   261	        """
   262	        for src in sources:
   263	            if not src:
   264	                continue
   265	            for key in ("n_queue", "queue_length", "lane_halting_number", "halting_number"):
   266	                val = src.get(key)
   267	                if val is not None:
   268	                    try:
   269	                        return max(0.0, float(val))
   270	                    except (TypeError, ValueError):
   271	                        continue
   272	        for src in sources:
   273	            if not src:
   274	                continue
   275	            gap = src.get("leader_gap")
   276	            lspd = src.get("leader_speed")
   277	            if gap is None or lspd is None:
   278	                continue
   279	            try:
   280	                gap_f, lspd_f = float(gap), float(lspd)
   281	            except (TypeError, ValueError):
   282	                continue
   283	            if math.isfinite(gap_f) and gap_f <= 30.0 and lspd_f <= 1.0:
   284	                return 1.0
   285	            return 0.0
   286	        return 0.0
   287	
   288	    @staticmethod
   289	    def _compute_heading(vx: float, vy: float, dx: float, dy: float) -> float:
   290	        """
   291	        Signed approach/recede indicator in [-1.0, 1.0].
   292	
   293	        dx, dy are the RSU-relative coordinates of the vehicle (vehicle - RSU), so the
   294	        vehicle->RSU direction is (-dx, -dy). Returns the normalized dot product of the
   295	        velocity vector with that direction: +1 approaching head-on, -1 receding,
   296	        0 when stopped or exactly at the RSU.
   297	        """
   298	        speed = math.hypot(vx, vy)
   299	        dist = math.hypot(dx, dy)
   300	        if speed < 1e-6 or dist < 1e-6:
   301	            return 0.0
   302	        cos_theta = (vx * (-dx) + vy * (-dy)) / (speed * dist)
   303	        return float(np.clip(cos_theta, -1.0, 1.0))

   305	    def vectorize(
   306	        self,
   307	        vehicle_node: Any,
   308	        rsu_node: Any,
   309	        current_time: float,
   310	        tls_info: Optional[Dict[str, Any]] = None,
   311	        cbr: float = 0.0,
   312	        n_active: int = 1,
   313	        n_queue: Optional[float] = None,
   314	    ) -> np.ndarray:
   315	        """
   316	        Vectorize node objects into a 17-dim normalized observation vector.
   317	
   318	        n_queue: explicit count of vehicles queued ahead in the same lane. When None
   319	        it is recovered from the TLS/dynamics feature dict (see _extract_queue_count).
   320	        """
   321	        vec = np.zeros(STATE_DIM, dtype=np.float32)
   322	        if vehicle_node is None or rsu_node is None:
   323	            return vec
   324	
   325	        # [0] Quality of the RSU's last prediction (see vectorize_from_dict).
   326	        vec[0] = norm_sq_error(float(getattr(vehicle_node, "last_pred_err", 0.0)))
   327	
   328	        # [1-3] Velocities and speed
   329	        vel = getattr(vehicle_node, "vel", (0.0, 0.0))
   330	        if hasattr(vehicle_node, "speed"):
   331	            spd_val = vehicle_node.speed() if callable(vehicle_node.speed) else vehicle_node.speed
   332	        else:
   333	            spd_val = math.hypot(vel[0], vel[1])
   334	        
   335	        vec[1] = np.clip(vel[0] / self.v_max, -1.0, 1.0)
   336	        vec[2] = np.clip(vel[1] / self.v_max, -1.0, 1.0)
   337	        vec[3] = np.clip(float(spd_val) / self.v_max, 0.0, 1.0)
   338	
   339	        # [4] Acceleration
   340	        accel = getattr(vehicle_node, "accel", 0.0)
   341	        vec[4] = np.clip(float(accel) / self.a_max, -1.0, 1.0)
   342	
   343	        # [5-7] Relative coordinates and Distance to RSU
   344	        pos = getattr(vehicle_node, "pos", (0.0, 0.0))
   345	        rsu_pos = getattr(rsu_node, "pos", (0.0, 0.0))
   346	        dx = float(pos[0] - rsu_pos[0])
   347	        dy = float(pos[1] - rsu_pos[1])
   348	        dist = math.hypot(dx, dy)
   349	        vec[5] = np.clip(dx / self.rsu_range, -1.0, 1.0)
   350	        vec[6] = np.clip(dy / self.rsu_range, -1.0, 1.0)
   351	        vec[7] = np.clip(dist / self.rsu_range, 0.0, 1.0)
   352	
   353	        # [8-12] TLS features (TraCI)
   354	        tls = tls_info or {}
   355	        if not tls and hasattr(vehicle_node, "_state_dict"):
   356	            st = vehicle_node._state_dict()
   357	            tls = st.get("tls_features", {})
   358	
   359	        state = str(tls.get("state", "g")).lower()
   360	        vec[8] = 1.0 if state in ["r", "red"] else 0.0
   361	        vec[9] = 1.0 if state in ["y", "yellow"] else 0.0
   362	        vec[10] = 1.0 if state in ["g", "green"] else 0.0
   363	
   364	        t_switch = float(tls.get("time_to_switch", 30.0))
   365	        vec[11] = np.clip(t_switch / 60.0, 0.0, 1.0)
   366	
   367	        d_stop = float(tls.get("dist_to_stopline", self.rsu_range))
   368	        vec[12] = np.clip(d_stop / self.rsu_range, 0.0, 1.0)
   369	
   370	        # [13-14] Network contention and measured channel occupancy
   371	        vec[13] = np.clip(float(n_active) / 100.0, 0.0, 1.0)
   372	        vec[14] = np.clip(float(cbr), 0.0, 1.0)
   373	
   374	        # [15] n_queue: vehicles queued ahead in the same lane (design S1)
   375	        if n_queue is None:
   376	            q_cnt = self._extract_queue_count(tls, getattr(vehicle_node, "__dict__", None))
   377	        else:
   378	            q_cnt = max(0.0, float(n_queue))
   379	        vec[15] = np.clip(q_cnt / self.queue_max, 0.0, 1.0)
   380	
   381	        # [16] heading: signed approach (+) / recede (-) indicator w.r.t. the RSU (design S1)
   382	        vec[16] = self._compute_heading(float(vel[0]), float(vel[1]), dx, dy)
   383	
   384	        return vec

   386	    def vectorize_from_dict(self, state_dict: Dict[str, Any], rsu_pos: Tuple[float, float] = (0.0, 0.0)) -> np.ndarray:
   387	        """
   388	        Convenience method to vectorize directly from a state dictionary.
   389	        """
   390	        vec = np.zeros(STATE_DIM, dtype=np.float32)
   391	        pos = state_dict.get("pos", (0.0, 0.0))
   392	        vel = state_dict.get("vel", (0.0, 0.0))
   393	        speed = state_dict.get("speed", math.hypot(vel[0], vel[1]))
   394	        accel = state_dict.get("accel", 0.0)
   395	        # [0] Quality of the RSU's last prediction for this vehicle.
   396	        #
   397	        # This slot used to hold normalized age. Under the SMDP formulation age
   398	        # is structurally zero at every decision epoch -- the RSU decides right
   399	        # after an update lands -- so the feature reached the policy as a
   400	        # constant and carried nothing. Measured: 1 unique value over a full run.
   401	        #
   402	        # What the RSU does know at that instant, for free, is how wrong its
   403	        # dead-reckoned belief turned out to be just before the report arrived.
   404	        # That is a direct read on how far this particular vehicle can be trusted
   405	        # to stay predictable, which is exactly what choosing Delta needs.
   406	        vec[0] = norm_sq_error(float(state_dict.get("last_pred_err", 0.0)))
   407	        vec[1] = np.clip(vel[0] / self.v_max, -1.0, 1.0)
   408	        vec[2] = np.clip(vel[1] / self.v_max, -1.0, 1.0)
   409	        vec[3] = np.clip(float(speed) / self.v_max, 0.0, 1.0)
   410	        vec[4] = np.clip(float(accel) / self.a_max, -1.0, 1.0)
   411	
   412	        dx = float(pos[0] - rsu_pos[0])
   413	        dy = float(pos[1] - rsu_pos[1])
   414	        dist = state_dict.get("dist_to_rsu", math.hypot(dx, dy))
   415	        vec[5] = np.clip(dx / self.rsu_range, -1.0, 1.0)
   416	        vec[6] = np.clip(dy / self.rsu_range, -1.0, 1.0)
   417	        vec[7] = np.clip(float(dist) / self.rsu_range, 0.0, 1.0)
   418	
   419	        tls = state_dict.get("tls_features", state_dict)
   420	        state = str(tls.get("state", "g")).lower()
   421	        vec[8] = 1.0 if state in ["r", "red"] else 0.0
   422	        vec[9] = 1.0 if state in ["y", "yellow"] else 0.0
   423	        vec[10] = 1.0 if state in ["g", "green"] else 0.0
   424	        vec[11] = np.clip(float(tls.get("time_to_switch", 30.0)) / 60.0, 0.0, 1.0)
   425	        vec[12] = np.clip(float(tls.get("dist_to_stopline", self.rsu_range)) / self.rsu_range, 0.0, 1.0)
   426	
   427	        vec[13] = np.clip(float(state_dict.get("n_active", 1)) / 100.0, 0.0, 1.0)
   428	        vec[14] = np.clip(float(state_dict.get("cbr", 0.0)), 0.0, 1.0)
   429	
   430	        # [15] n_queue: vehicles queued ahead in the same lane (design S1).
   431	        # Prefer an explicit count on the state dict, then the TLS/dynamics feature
   432	        # dict, then the leader-vehicle fallback.
   433	        q_cnt = self._extract_queue_count(state_dict, tls)
   434	        vec[15] = np.clip(q_cnt / self.queue_max, 0.0, 1.0)
   435	
   436	        # [16] heading: signed approach (+) / recede (-) indicator w.r.t. the RSU (design S1)
   437	        vec[16] = self._compute_heading(float(vel[0]), float(vel[1]), dx, dy)
   438	
   439	        return vec

   442	class ActionDecoder:
   443	    """
   444	    Hybrid Action Space Decoder.
   445	    
   446	    Decodes raw model logits or tensor outputs into a concrete uplink grant 3-tuple:
   447	    (Delta in [0.1, 45.0]s, ch in {0, 1, 2, 3}, power in [10.0, 23.0]dBm).
   448	
   449	    These bounds are the user-approved design (Conversation.md S2) and are the SINGLE
   450	    SOURCE OF TRUTH for the action space. Rationale for the record:
   451	      * p_max = 23 dBm -- 3GPP TS 36.101/38.101 power-class-3 UE maximum transmit power.
   452	      * delta_min = 0.1 s -- ETSI EN 302 637-2 CAM minimum generation interval
   453	        (T_GenCamMin); generating faster than this is not standards-compliant.
   454	      * delta_max = 45.0 s -- worst-case standstill duration in the actual SUMO
   455	        scenario (generated.net.xml tlLogic: green 42 s + yellow 3 s => 45 s red per
   456	        approach), i.e. the longest interval over which a stopped vehicle's mobility
   457	        state provably need not be refreshed.
   458	
   459	    Delta mapping is GEOMETRIC, not linear. The Delta range spans a factor of 450, so a
   460	    linear interpolation of sigmoid(logit) would need u ~= 0.0089 just to emit 0.5 s and
   461	    would destroy all resolution in the short-interval regime. Instead:
   462	
   463	        u     = sigmoid(raw_delta)                              in [0, 1]
   464	        delta = delta_min * (delta_max / delta_min) ** u        (inverse: u = log(d/d_min)/log(d_max/d_min))
   465	
   466	    which gives uniform *relative* resolution across u. Power stays linear: dBm is
   467	    already a logarithmic unit.
   468	
   469	    Downstream code must read delta_min/delta_max/p_min/p_max off the decoder
   470	    instance rather than duplicating the literals.
   471	    """
   472	
   473	    def __init__(
   474	        self,
   475	        num_channels: int = 4,
   476	        delta_min: float = DELTA_MIN,
   477	        delta_max: float = DELTA_MAX,
   478	        p_min: float = P_MIN,
   479	        p_max: float = P_MAX,
   480	    ) -> None:
   481	        self.num_channels = int(num_channels)
   482	        self.delta_min = float(delta_min)
   483	        self.delta_max = float(delta_max)
   484	        self.p_min = float(p_min)
   485	        self.p_max = float(p_max)
   486	        # log(delta_max / delta_min): the geometric span used by the Delta mapping.
   487	        # Guarded so a degenerate delta_min == delta_max decoder still works.
   488	        if self.delta_min > 0.0 and self.delta_max > self.delta_min:
   489	            self._log_delta_ratio = math.log(self.delta_max / self.delta_min)
   490	        else:
   491	            self._log_delta_ratio = 0.0
   492	
   493	    def delta_from_unit(self, u: float) -> float:
   494	        """Geometric map u in [0, 1] -> Delta in [delta_min, delta_max]."""
   495	        u = min(max(float(u), 0.0), 1.0)
   496	        if self._log_delta_ratio <= 0.0:
   497	            # Degenerate range: fall back to linear interpolation.
   498	            return self.delta_min + u * (self.delta_max - self.delta_min)
   499	        if u <= 0.0:
   500	            return self.delta_min
   501	        if u >= 1.0:
   502	            return self.delta_max
   503	        return self.delta_min * math.exp(u * self._log_delta_ratio)
   504	
   505	    def unit_from_delta(self, delta: float) -> float:
   506	        """Inverse geometric map Delta -> u in [0, 1]."""
   507	        d = min(max(float(delta), self.delta_min), self.delta_max)
   508	        if self._log_delta_ratio <= 0.0:
   509	            return (d - self.delta_min) / max(1e-6, self.delta_max - self.delta_min)
   510	        return math.log(d / self.delta_min) / self._log_delta_ratio
   511	
   512	    @staticmethod
   513	    def _sigmoid(x: float) -> float:
   514	        if x < -50.0:
   515	            return 0.0
   516	        if x > 50.0:
   517	            return 1.0
   518	        return 1.0 / (1.0 + math.exp(-x))
   519	
   520	    @staticmethod
   521	    def _logit(p: float) -> float:
   522	        p_clamped = min(max(p, 1e-6), 1.0 - 1e-6)
   523	        return math.log(p_clamped / (1.0 - p_clamped))
   524	
   525	    def decode_action(self, raw_action: Any) -> Tuple[float, int, float]:
   526	        """
   527	        Decodes raw action into (delta_s, channel_idx, power_dbm).
   528	        """
   529	        if isinstance(raw_action, dict):
   530	            raw_delta = raw_action.get("delta", raw_action.get("delta_raw", 0.0))
   531	            raw_ch = raw_action.get("ch", raw_action.get("channel_idx", 0))
   532	            raw_p = raw_action.get("power", raw_action.get("tx_power_dbm", 0.0))
   533	        elif isinstance(raw_action, (list, tuple, np.ndarray, torch.Tensor)):
   534	            if isinstance(raw_action, torch.Tensor):
   535	                raw_action = raw_action.detach().cpu().numpy().flatten()
   536	            raw_list = list(raw_action)
   537	            if len(raw_list) >= 3:
   538	                raw_delta, raw_ch, raw_p = raw_list[0], raw_list[1], raw_list[2]
   539	            elif len(raw_list) == 2:
   540	                raw_delta, raw_ch, raw_p = raw_list[0], 0, raw_list[1]
   541	            elif len(raw_list) == 1:
   542	                raw_delta, raw_ch, raw_p = raw_list[0], 0, 0.0
   543	            else:
   544	                raw_delta, raw_ch, raw_p = 0.0, 0, 0.0
   545	        else:
   546	            raw_delta, raw_ch, raw_p = 0.0, 0, 0.0
   547	
   548	        # Continuous delta mapping [delta_min, delta_max]
   549	        sig_d = self._sigmoid(float(raw_delta))
   550	        delta = self.delta_min + sig_d * (self.delta_max - self.delta_min)
   551	
   552	        # Discrete channel mapping {0..num_channels-1}
   553	        ch = int(round(float(raw_ch))) % self.num_channels
   554	
   555	        # Continuous power mapping [p_min, p_max]
   556	        sig_p = self._sigmoid(float(raw_p))
   557	        power = self.p_min + sig_p * (self.p_max - self.p_min)
   558	
   559	        return (float(delta), int(ch), float(power))
   560	
   561	    def encode_action(self, delta: float, ch: int, power: float) -> np.ndarray:
   562	        """
   563	        Inverse logit encoding from (delta, ch, power) to raw action values.
   564	        """
   565	        norm_d = (delta - self.delta_min) / max(1e-6, self.delta_max - self.delta_min)
   566	        norm_p = (power - self.p_min) / max(1e-6, self.p_max - self.p_min)
   567	        raw_d = self._logit(norm_d)
   568	        raw_p = self._logit(norm_p)
   569	        return np.array([raw_d, float(ch), raw_p], dtype=np.float32)
   570	
   571	
   572	class RetrospectiveReplayBuffer:
   573	    """
   574	    SMDP Retrospective Replay Buffer with variable-interval discount gamma^Delta support.
   575	    """
   576	
   577	    def __init__(self, capacity: int = 10000, gamma: float = 0.99) -> None:
   578	        self.capacity = int(capacity)
   579	        self.gamma = float(gamma)
   580	        self.buffer: List[Dict[str, Any]] = []
   581	        self.position = 0
   582	
   583	    def push(
   584	        self,
   585	        state: Union[np.ndarray, torch.Tensor, List[float]],
   586	        action: Union[np.ndarray, torch.Tensor, List[float], Tuple[float, ...]],
   587	        reward: float,
   588	        next_state: Union[np.ndarray, torch.Tensor, List[float]],
   589	        done: bool,
   590	        delta_t: float,
   591	        action_idx: Optional[int] = None,
   592	    ) -> None:
   593	        """
   594	        Push a transition tuple (s, a, r, s', done, delta_t) into buffer.
   595	
   596	        action_idx (optional): the combined *discrete* action index that a
   597	        discrete-action agent (e.g. DuelingQAoI, whose grid is
   598	        interval_idx * num_channels + channel_idx) actually selected. It is
   599	        stored verbatim so that credit assignment in update() targets the true
   600	        action instead of a lossy reconstruction from the decoded continuous
   601	        action. Agents with purely continuous / factorized action heads simply
   602	        leave it as None; sample() then omits the "action_idx" key entirely,
   603	        so batch contents are unchanged for every other baseline.
   604	        """
   605	        s_arr = np.array(state, dtype=np.float32) if not isinstance(state, np.ndarray) else state.astype(np.float32)
   606	        if isinstance(action, torch.Tensor):
   607	            a_arr = action.detach().cpu().numpy().astype(np.float32)
   608	        elif isinstance(action, (list, tuple)):
   609	            a_arr = np.array(action, dtype=np.float32)
   610	        else:
   611	            a_arr = np.array(action, dtype=np.float32)
   612	
   613	        ns_arr = np.array(next_state, dtype=np.float32) if not isinstance(next_state, np.ndarray) else next_state.astype(np.float32)
   614	
   615	        item = {
   616	            "state": s_arr,
   617	            "action": a_arr,
   618	            "reward": float(reward),
   619	            "next_state": ns_arr,
   620	            "done": float(done),
   621	            "delta_t": float(delta_t),
   622	            "action_idx": None if action_idx is None else int(action_idx),
   623	        }
   624	
   625	        if len(self.buffer) < self.capacity:
   626	            self.buffer.append(item)
   627	        else:
   628	            self.buffer[self.position] = item
   629	        self.position = (self.position + 1) % self.capacity
   630	
   631	    def sample(self, batch_size: int) -> Dict[str, torch.Tensor]:
   632	        """
   633	        Sample a random batch of transitions as PyTorch Tensors.
   634	        Raises ValueError if buffer is empty.
   635	        """
   636	        if len(self.buffer) == 0:
   637	            raise ValueError("Cannot sample from an empty buffer.")
   638	
   639	        batch_size = min(batch_size, len(self.buffer))
   640	        indices = np.random.choice(len(self.buffer), batch_size, replace=False)
   641	        batch = [self.buffer[i] for i in indices]
   642	
   643	        states = np.array([b["state"] for b in batch], dtype=np.float32)
   644	        actions = np.array([b["action"] for b in batch], dtype=np.float32)
   645	        rewards = np.array([[b["reward"]] for b in batch], dtype=np.float32)
   646	        next_states = np.array([b["next_state"] for b in batch], dtype=np.float32)
   647	        dones = np.array([[b["done"]] for b in batch], dtype=np.float32)
   648	        delta_ts = np.array([[b["delta_t"]] for b in batch], dtype=np.float32)
   649	        discounts = np.power(self.gamma, delta_ts).astype(np.float32)
   650	
   651	        out = {
   652	            "state": torch.from_numpy(states),
   653	            "action": torch.from_numpy(actions),
   654	            "reward": torch.from_numpy(rewards),
   655	            "next_state": torch.from_numpy(next_states),
   656	            "done": torch.from_numpy(dones),
   657	            "delta_t": torch.from_numpy(delta_ts),
   658	            "discount": torch.from_numpy(discounts),
   659	        }
   660	
   661	        # Optional discrete action indices. Emitted only when EVERY sampled
   662	        # transition carries one, so batches for continuous-action baselines
   663	        # keep exactly the legacy key set / tensor shapes.
   664	        idx_list = [b.get("action_idx") for b in batch]
   665	        if len(idx_list) > 0 and all(i is not None for i in idx_list):
   666	            out["action_idx"] = torch.from_numpy(np.array(idx_list, dtype=np.int64))
   667	
   668	        return out
   669	
   670	    def is_ready(self, batch_size: int) -> bool:
   671	        return len(self.buffer) >= batch_size
   672	
   673	    def clear(self) -> None:
   674	        self.buffer.clear()
   675	        self.position = 0
   676	
   677	    def __len__(self) -> int:
   678	        return len(self.buffer)
